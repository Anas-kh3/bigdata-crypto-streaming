## 5) Normalized Price (base = 100 at start) — BTC & ETH
data =
  from(bucket: "crypto")
    |> range(start: -6h)
    |> filter(fn: (r) => r._measurement == "crypto_1min")
    |> filter(fn: (r) => r._field == "avg_price")
    |> filter(fn: (r) => r.symbol == "btc" or r.symbol == "eth")

base =
  data
    |> group(columns: ["symbol"])
    |> first()
    |> keep(columns: ["symbol", "_value"])
    |> rename(columns: {_value: "base"})

join(tables: {d: data, b: base}, on: ["symbol"])
  |> map(fn: (r) => ({ r with _value: (r._value / r.base) * 100.0 }))
## 6) BTC/ETH ratio (robuste)
btc =
  from(bucket: "crypto")
    |> range(start: -6h)
    |> filter(fn: (r) => r._measurement == "crypto_1min")
    |> filter(fn: (r) => r._field == "avg_price")
    |> filter(fn: (r) => r.symbol == "btc")
    |> aggregateWindow(every: 1m, fn: mean, createEmpty: false)
    |> keep(columns: ["_time", "_value"])
    |> rename(columns: {_value: "btc"})

eth =
  from(bucket: "crypto")
    |> range(start: -6h)
    |> filter(fn: (r) => r._measurement == "crypto_1min")
    |> filter(fn: (r) => r._field == "avg_price")
    |> filter(fn: (r) => r.symbol == "eth")
    |> aggregateWindow(every: 1m, fn: mean, createEmpty: false)
    |> keep(columns: ["_time", "_value"])
    |> rename(columns: {_value: "eth"})

join(tables: {b: btc, e: eth}, on: ["_time"])
  |> map(fn: (r) => ({ _time: r._time, _value: r.btc / r.eth }))
